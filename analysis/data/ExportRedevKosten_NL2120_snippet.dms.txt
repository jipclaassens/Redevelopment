{

	attribute<Eur_Ha> Low_Faktondatabase (AdminDomain) := /VariantData/BAU/Suitability/src/Wonen/Grondproductiekosten/Results/Totaal/Low_Faktondatabase

	, StorageName = "=Dir+'/High_Exploitatieplannen.tif'", StorageType = "gdalwrite.grid", IntegrityCheck = "AndersBouwenAan && Prijspeil2023";

	, Descr = "Activeer dit item om de zes grondproductiekosten-grids weg te schrijven.";
